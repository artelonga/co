### `e2e/wave-2/co-98-sidebar-tree.spec.ts` (CO-98 acceptance #7)

```ts
test('logged-in user sees timeline trio nested under template in sidebar', async ({ page }) => {
    await loginAsAdmin(page);                      // password-login fixture
    await page.goto(`${BASE_URL}/co?u=template`);
    await expect(page.locator('.sidebar-universes')).toBeVisible();
    // template parent is rendered with chevron
    await expect(page.locator('[data-universe="template"] .sidebar-universe-chevron'))
        .toContainText(/[▾▸]/);
    // expanded by default since a child is active or per default
    await page.locator('[data-toggle="template"]').click();
    await page.locator('[data-toggle="template"]').click();  // re-expand
    // children appear with indent
    for (const child of ['tempo', 'humanity', 'universo']) {
        const el = page.locator(`[data-universe="${child}"]`);
        await expect(el).toBeVisible();
        const padLeft = await el.evaluate(n => parseInt(getComputedStyle(n).paddingLeft, 10));
        expect(padLeft).toBeGreaterThan(20);  // indented vs top-level
    }
});
```

### `e2e/wave-2/co-107-mermaid.spec.ts` (CO-107 acceptance #6)

```ts
test('template home renders mermaid svg', async ({ page }) => {
    await page.goto(`${BASE_URL}/co?u=template`);
    // Wait for the home view
    await page.waitForSelector('.universe-home-md', { timeout: 10000 });
    // Wait for lazy-loaded mermaid bundle to replace the code fence with svg
    const svg = page.locator('.universe-home-md .co-mermaid svg');
    await expect(svg).toBeVisible({ timeout: 8000 });
    // Sanity: contains the trio node labels
    const text = await svg.textContent();
    expect(text).toContain('Tempo');
    expect(text).toContain('Universo');
    expect(text).toContain('Humanidade');
});

test('universes without mermaid blocks do not load mermaid bundle', async ({ page }) => {
    const requests: string[] = [];
    page.on('request', r => { if (r.url().includes('mermaid')) requests.push(r.url()); });
    await page.goto(`${BASE_URL}/co?u=quilomboaraucaria`);  // assume no mermaid in its index.md
    await page.waitForTimeout(2000);
    expect(requests).toEqual([]);
});
```

### `e2e/wave-2/co-99-onboarding.spec.ts` (CO-99 acceptance)

```ts
test('first-time anonymous template visit shows 3-step banner and dismisses', async ({ page, context }) => {
    await context.clearCookies();
    await page.setViewportSize({ width: 1024, height: 768 });
    await page.goto(`${BASE_URL}/co?u=template`);
    const banner = page.locator('#onboarding-banner');
    await expect(banner).toBeVisible({ timeout: 5000 });
    await expect(banner.locator('#onboarding-step-indicator')).toHaveText('1 / 3');
    await banner.locator('#onboarding-next').click();
    await expect(banner.locator('#onboarding-step-indicator')).toHaveText('2 / 3');
    await banner.locator('#onboarding-next').click();
    await expect(banner.locator('#onboarding-step-indicator')).toHaveText('3 / 3');
    await banner.locator('#onboarding-next').click();          // "Concluir"
    await expect(banner).toBeHidden();
    // Cookie set
    const cookies = await context.cookies();
    expect(cookies.find(c => c.name === 'co_onboarded')?.value).toBe('1');
    // Reload — banner should NOT reappear
    await page.reload();
    await expect(banner).toBeHidden();
});

test('mobile viewport (<720px) suppresses banner entirely', async ({ page, context }) => {
    await context.clearCookies();
    await page.setViewportSize({ width: 600, height: 800 });
    await page.goto(`${BASE_URL}/co?u=template`);
    await page.waitForTimeout(2000);
    await expect(page.locator('#onboarding-banner')).toBeHidden();
});

test('logged-in user does NOT see banner on template', async ({ page, context }) => {
    await context.clearCookies();
    await loginAsAdmin(page);
    await page.goto(`${BASE_URL}/co?u=template`);
    await page.waitForTimeout(2000);
    await expect(page.locator('#onboarding-banner')).toBeHidden();
});
```

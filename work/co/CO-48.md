## tables.yaml format

```yaml
tables:
  - name: users
    description: Authenticated user accounts
    columns:
      - name: id
        type: TEXT
        primary_key: true
        description: nanoid user ID, format usr_xxxxx
      - name: email
        type: TEXT
        unique: true
        nullable: false
      - name: display_name
        type: TEXT
        nullable: true
      - name: senha_hash
        type: TEXT
        description: Argon2 hash
      - name: tier
        type: TEXT
        default: player
        enum: [player, admin]
      - name: created_at
        type: TEXT
        format: rfc3339
    indexes:
      - columns: [email]
        unique: true

  - name: universes
    description: Multi-tenant workspaces
    columns:
      - name: key
        type: TEXT
        primary_key: true
      - name: name
      - name: description
      - name: owner_id
        type: TEXT
        references: users.id
        nullable: true
        description: NULL for system-owned (template)
      - name: is_template
        type: INTEGER
        default: 0
      - name: is_public
        type: INTEGER
        default: 0
      - name: theme_preset
        type: TEXT
        default: scholarly
      - name: layout
        type: TEXT
        default: board
      - name: content_count
        type: INTEGER
        default: 0
    relations:
      - kind: one_to_many
        target: entries
        on: universe_key

  - name: entries
    description: Universal content storage (tasks, projects, pages, events, etc.)
    columns:
      - name: path
        type: TEXT
      - name: universe_key
        type: TEXT
        references: universes.key
      - name: entry_type
        type: TEXT
        enum: [project, task, page, event, message, comment, clip]
      - name: title
        type: TEXT
      - name: frontmatter_json
        type: TEXT
      - name: body
        type: TEXT
      - name: body_hash
        type: TEXT
      - name: created_at
        type: TEXT
      - name: updated_at
        type: TEXT
    primary_key: [universe_key, path]
    indexes:
      - columns: [universe_key, entry_type]
      - columns: [universe_key, updated_at]

  # ... document ALL tables from migrations v1-v13
```

## frontmatter.yaml format

```yaml
content_types:
  - type: task
    description: A unit of work in a project
    required: [type, id, title, status, project]
    fields:
      type: { value: task }
      id: { type: integer }
      title: { type: string, max: 200 }
      status: { type: enum, values: [todo, in_progress, in_review, done] }
      priority: { type: enum, values: [low, medium, high, critical] }
      project: { type: string, references: project.key }
      parent: { type: integer, references: task.id, nullable: true }
      tags: { type: array, items: string }
      assignee: { type: string, nullable: true }
      due: { type: date, nullable: true }
      created: { type: datetime }
      modified: { type: datetime }

  - type: project
    description: Container for tasks
    fields:
      type: { value: project }
      key: { type: string, pattern: '[A-Z]+' }
      title: { type: string }
      status: { type: enum, values: [active, archived] }

  - type: page
    description: Free-form markdown content
    fields:
      type: { value: page }
      slug: { type: string }
      title: { type: string }
      order: { type: integer, default: 0 }
      parent: { type: string, nullable: true, description: parent slug for hierarchy }
      tags: { type: array }

  # ... event, message, comment, clip
```

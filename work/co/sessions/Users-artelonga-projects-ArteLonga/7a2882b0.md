```js
function computeFaixaPreco(s) {
  if (s.faixaPreco) return s.faixaPreco;        // override vence
  if (!s.hoursLow || !s.hoursHigh) return null;
  const ativos = s.responsavel.filter(h => !isInactive(h));
  if (!ativos.length) return null;
  const rates = ativos.map(h => (get(h)?.hourlyRate) || DEFAULT_HOURLY_RATE);
  const low  = s.hoursLow  * Math.min(...rates);
  const high = s.hoursHigh * Math.max(...rates);
  const fmt = n => "R$ " + n.toLocaleString("pt-BR");
  const suffix = s.recurring ? "/mês" : "";
  if (low === high) return `${fmt(low)}${suffix}`;
  return `${fmt(low)} – ${fmt(high)}${suffix}`;
}
```

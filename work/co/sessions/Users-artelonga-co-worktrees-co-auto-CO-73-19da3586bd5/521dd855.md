     pub fn derive_master_key(password: &str, salt: &[u8]) -> anyhow::Result<[u8; KEY_LEN]> {
         let mut full_salt = b"co:salt:v1:".to_vec();
         full_salt.extend_from_slice(salt);
         let argon = Argon2::new(Algorithm::Argon2id, Version::V0x13, argon2_params());
         let mut out = [0u8; KEY_LEN];
         argon.hash_password_into(password.as_bytes(), &full_salt, &mut out)
             .map_err(|e| anyhow::anyhow!("argon2id derivation failed: {e}"))?;
         Ok(out)
     }
     fn hkdf_expand(ikm: &[u8], info: &[u8]) -> [u8; KEY_LEN] {
         let hk = Hkdf::<Sha256>::new(None, ikm);
         let mut out = [0u8; KEY_LEN];
         hk.expand(info, &mut out).expect("32 bytes fits Sha256");
         out
     }
     pub fn derive_auth_key(master_key: &[u8]) -> [u8; KEY_LEN] {
         hkdf_expand(master_key, b"co:auth")
     }
     pub fn auth_key_for(password: &str, username: &str) -> anyhow::Result<String> {
         let master = derive_master_key(password, username.as_bytes())?;
         let auth = derive_auth_key(&master);
         Ok(hex::encode(auth))
